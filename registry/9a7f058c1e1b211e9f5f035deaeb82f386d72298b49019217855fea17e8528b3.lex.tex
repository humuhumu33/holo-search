  body: (match fuel
    | Nat.zero => (match nodes | List.nil => (ctor Option.some [Facts] facts) | List.cons node rest => (ctor Option.none [Facts]))
    | Nat.succ less => (match nodes
      | List.nil => (ctor Option.some [Facts] facts)
      | List.cons node rest => (match node
        | Node.Heading words => (collect less rest facts)
        | Node.Action words => (collect less rest facts)
        | Node.Field field => (collect less rest (noted facts (project field id) false true))
        | Node.Choice choice => (collect less rest (noted facts (project choice id) false (offers (project choice initial) (project choice options))))
        | Node.Output key => (collect less rest (noted facts key true true))
        | Node.Row children => (match (collect less children facts)
          | Option.none => (ctor Option.none [Facts])
          | Option.some inner => (collect less rest inner))
        | Node.Stack children => (match (collect less children facts)
          | Option.none => (ctor Option.none [Facts])
          | Option.some inner => (collect less rest inner))))),
  executable: true,
  recursive_argument: fuel
}
